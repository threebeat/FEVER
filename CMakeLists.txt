cmake_minimum_required(VERSION 3.20)
project(FEVER VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
add_library(incident STATIC src/effects.cpp src/behavior.cpp src/game.cpp src/scene.cpp src/audio.cpp src/ui.cpp)
target_include_directories(incident PUBLIC src)
if(WIN32)
    add_executable(FEVER WIN32 src/windows.cpp src/renderer.cpp)
    target_link_libraries(FEVER PRIVATE incident d3d11 dxgi d3dcompiler windowscodecs ole32 uuid user32 winmm gdi32)
    if(MSVC)
        set_property(TARGET FEVER incident PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_custom_command(TARGET FEVER POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${PROJECT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:FEVER>/assets"
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:FEVER>/docs"
        COMMAND ${CMAKE_COMMAND} -E copy "${PROJECT_SOURCE_DIR}/docs/DIALOGUE.md" "$<TARGET_FILE_DIR:FEVER>/docs/DIALOGUE.md")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(incident_tests tests/incident.cpp)
    target_link_libraries(incident_tests PRIVATE incident)
    if(MSVC)
        set_property(TARGET incident_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_test(NAME incident COMMAND incident_tests)
    set_tests_properties(incident PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
endif()
